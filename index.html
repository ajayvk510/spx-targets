function updateSPX() { // ============================== // SETTINGS // ============================== const SYMBOL = "%5EGSPC"; const YAHOO_URL = "https://query1.finance.yahoo.com/v8/finance/chart/" + SYMBOL + "?interval=1d&range=5d"; const SHEET_NAME = "SPX"; const INNER = 40; // interior width of the box — widen if numbers ever run longer const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } try { // ============================== // GET YAHOO DATA // ============================== const response = UrlFetchApp.fetch(YAHOO_URL, { muteHttpExceptions: true, headers: { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36", "Accept": "application/json" } }); const code = response.getResponseCode(); const body = response.getContentText(); if (code !== 200) { throw new Error("Yahoo Finance HTTP " + code + ": " + body.slice(0, 300)); } const json = JSON.parse(body); if (json.chart.error) { throw new Error("Yahoo API error: " + JSON.stringify(json.chart.error)); } const result = json.chart.result && json.chart.result[0]; if (!result || !result.meta) { throw new Error("Unexpected Yahoo response shape: " + body.slice(0, 300)); } const meta = result.meta; // ============================== // PREVIOUS CLOSE // ============================== const previousClose = meta.chartPreviousClose || meta.previousClose; if (typeof previousClose !== "number") { throw new Error("Missing previousClose in Yahoo response"); } // ============================== // TODAY'S OPEN // ============================== const timestamps = result.timestamp || []; const opens = (result.indicators && result.indicators.quote && result.indicators.quote[0] && result.indicators.quote[0].open) || []; const timezone = meta.exchangeTimezoneName || "America/New_York"; const refTime = meta.regularMarketTime ? new Date(meta.regularMarketTime * 1000) : new Date(); const todayKey = refTime.toLocaleDateString("en-CA", { timeZone: timezone }); let todayOpen = null; for (let i = timestamps.length - 1; i >= 0; i--) { const key = new Date(timestamps[i] * 1000).toLocaleDateString("en-CA", { timeZone: timezone }); if (key === todayKey && typeof opens[i] === "number") { todayOpen = opens[i]; break; } } if (todayOpen === null || todayOpen === undefined) { todayOpen = previousClose; } // ============================== // CURRENT SPX / TARGETS // ============================== const currentPrice = meta.regularMarketPrice; const openChangePct = ((todayOpen - previousClose) / previousClose) * 100; const targets = { pcUp02: previousClose * 1.002, pcDown02: previousClose * 0.998, pcUp05: previousClose * 1.005, pcDown05: previousClose * 0.995, openUp05: todayOpen * 1.005, openDown05: todayOpen * 0.995 }; // ============================== // BOX BUILDING HELPERS // ============================== function fmtPrice(n) { return n.toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 }); } function fmtPct(n) { const sign = n > 0 ? "+" : (n < 0 ? "" : ""); return sign + n.toFixed(2) + "%"; } function row(label, value) { const left = " " + label; const gapLen = Math.max(1, INNER - left.length - value.length); const text = (left + " ".repeat(gapLen) + value).padEnd(INNER, " ").slice(0, INNER); return "│" + text + "│"; } function centered(text) { const pad = INNER - text.length; const l = Math.floor(pad / 2); const r = pad - l; return "│" + " ".repeat(Math.max(l, 0)) + text + " ".repeat(Math.max(r, 0)) + "│"; } function blank() { return "│" + " ".repeat(INNER) + "│"; } const top = "┌" + "─".repeat(INNER) + "┐"; const div = "├" + "─".repeat(INNER) + "┤"; const bottom = "└" + "─".repeat(INNER) + "┘"; const lastUpdated = Utilities.formatDate(new Date(), timezone, "h:mm a"); const lines = [ top, centered("SPX DAILY TARGETS"), div, blank(), row("Previous Close", fmtPrice(previousClose)), row("Today's Open", fmtPrice(todayOpen)), row("Open Change", fmtPct(openChangePct)), blank(), div, centered("TARGETS"), blank(), row("+0.20% Previous Close", fmtPrice(targets.pcUp02)), row("-0.20% Previous Close", fmtPrice(targets.pcDown02)), blank(), row("+0.50% Previous Close", fmtPrice(targets.pcUp05)), row("-0.50% Previous Close", fmtPrice(targets.pcDown05)), blank(), row("+0.50% Today's Open", fmtPrice(targets.openUp05)), row("-0.50% Today's Open", fmtPrice(targets.openDown05)), blank(), div, blank(), row("Current SPX", fmtPrice(currentPrice)), row("Last Updated", lastUpdated), bottom ]; // ============================== // WRITE TO SHEET // ============================== sheet.clear(); const rangeValues = lines.map(l => [l]); sheet.getRange(1, 1, rangeValues.length, 1).setValues(rangeValues); sheet.getRange(1, 1, rangeValues.length, 1) .setFontFamily("Courier New") .setFontSize(11) .setWrapStrategy(SpreadsheetApp.WrapStrategy.CLIP); sheet.setColumnWidth(1, 420); } catch (err) { sheet.clear(); sheet.getRange("A1").setValue("SPX update failed").setFontWeight("bold").setFontColor("#EA3943"); sheet.getRange("A2").setValue(String(err)); sheet.getRange("A3").setValue("Last attempt: " + new Date()); throw err; } }
