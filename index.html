function updateSPX() { // ============================== // SETTINGS // ============================== const SYMBOL = "%5EGSPC"; const YAHOO_URL = "https://query1.finance.yahoo.com/v8/finance/chart/" + SYMBOL + "?interval=1d&range=5d"; const SHEET_NAME = "SPX"; // ============================== // GET GOOGLE SHEET // ============================== const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } // ============================== // GET YAHOO DATA // ============================== const response = UrlFetchApp.fetch(YAHOO_URL, { muteHttpExceptions: true }); if (response.getResponseCode() !== 200) { throw new Error( "Yahoo Finance error: " + response.getResponseCode() ); } const json = JSON.parse(response.getContentText()); const result = json.chart.result[0]; const meta = result.meta; // ============================== // PREVIOUS CLOSE // ============================== const previousClose = meta.chartPreviousClose || meta.previousClose; // ============================== // TODAY'S OPEN // ============================== const timestamps = result.timestamp || []; const opens = result.indicators.quote[0].open || []; const timezone = meta.exchangeTimezoneName || "America/New_York"; const todayKey = new Date(meta.regularMarketTime * 1000) .toLocaleDateString( "en-CA", { timeZone: timezone } ); let todayOpen = null; for ( let i = timestamps.length - 1; i >= 0; i-- ) { const key = new Date(timestamps[i] * 1000) .toLocaleDateString( "en-CA", { timeZone: timezone } ); if ( key === todayKey && typeof opens[i] === "number" ) { todayOpen = opens[i]; break; } } // ============================== // FALLBACK // ============================== if ( todayOpen === null || todayOpen === undefined ) { todayOpen = previousClose; } // ============================== // CURRENT SPX // ============================== const currentPrice = meta.regularMarketPrice; // ============================== // OPEN CHANGE % // ============================== const openChangePct = ((todayOpen - previousClose) / previousClose) * 100; // ============================== // TARGETS // ============================== const up02 = previousClose * 1.002; const up05 = previousClose * 1.005; const down02 = previousClose * 0.998; const down05 = previousClose * 0.995; // ============================== // CLEAR SHEET // ============================== sheet.clear(); // ============================== // TITLE // ============================== sheet.getRange("A1:B1") .merge(); sheet.getRange("A1") .setValue("SPX DAILY TARGETS") .setFontWeight("bold") .setFontSize(16) .setHorizontalAlignment("center"); // ============================== // DATA // ============================== const values = [ ["Previous Close", previousClose], ["Today's Open", todayOpen], ["Open Change %", openChangePct / 100], ["", ""], ["TARGETS", ""], ["+0.20%", up02], ["+0.50%", up05], ["-0.20%", down02], ["-0.50%", down05], ["", ""], ["Current SPX", currentPrice], ["Last Updated", new Date()] ]; sheet .getRange(2, 1, values.length, 2) .setValues(values); // ============================== // FORMATTING // ============================== sheet .getRange("B2:B3") .setNumberFormat("#,##0.00"); sheet .getRange("B4") .setNumberFormat("0.00%"); sheet .getRange("B7:B10") .setNumberFormat("#,##0.00"); sheet .getRange("B12") .setNumberFormat("#,##0.00"); sheet .getRange("B13") .setNumberFormat( "yyyy-mm-dd hh:mm:ss" ); // ============================== // TARGET FORMATTING // ============================== sheet .getRange("A7:B8") .setFontColor("#16C784"); sheet .getRange("A9:B10") .setFontColor("#EA3943"); // ============================== // HEADER FORMATTING // ============================== sheet .getRange("A6:B6") .setFontWeight("bold") .setHorizontalAlignment("center"); // ============================== // COLUMN WIDTH // ============================== sheet.setColumnWidth(1, 180); sheet.setColumnWidth(2, 130); // ============================== // BORDERS // ============================== sheet .getRange("A1:B13") .setBorder( true, true, true, true, true, true ); // ============================== // ALIGNMENT // ============================== sheet .getRange("B2:B13") .setHorizontalAlignment("right"); }
